% keybox (红) — 核心概念/关键公式
\newtcolorbox{keybox}[1][]{
  colback=diffold, colframe=accent, title=#1,
  fonttitle=\bfseries\sffamily\color{accent!80!black}, breakable,
  boxrule=0.5pt, left=8pt, right=8pt, top=6pt, bottom=6pt
}

}
